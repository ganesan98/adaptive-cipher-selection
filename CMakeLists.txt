cmake_minimum_required(VERSION 3.20)
project(adaptive_cipher VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenSSL 3 REQUIRED COMPONENTS Crypto)
add_library(acs INTERFACE)
target_include_directories(acs INTERFACE include)
target_link_libraries(acs INTERFACE OpenSSL::Crypto)
if(WIN32)
  target_link_libraries(acs INTERFACE ws2_32)
endif()
if(MSVC)
  target_compile_options(acs INTERFACE /W4)
else()
  target_compile_options(acs INTERFACE -Wall -Wextra -Wpedantic)
endif()
foreach(name client server)
  add_executable(acs_${name} src/${name}.cpp)
  target_link_libraries(acs_${name} PRIVATE acs)
endforeach()
enable_testing()
add_executable(acs_tests tests/core_test.cpp)
target_link_libraries(acs_tests PRIVATE acs)
add_test(NAME core COMMAND acs_tests ${CMAKE_SOURCE_DIR}/models/policy.tree)
add_executable(policy_test tests/policy_test.cpp)
target_link_libraries(policy_test PRIVATE acs)
add_test(NAME model_parity COMMAND policy_test ${CMAKE_SOURCE_DIR}/models/policy.tree ${CMAKE_SOURCE_DIR}/models/parity.csv)
